----QUESTION 4: Trigger – Update Account Balance (20 Marks)---

CREATE TRIGGER trg_AfterInsert_Transactions
ON Transactions
AFTER INSERT
AS
BEGIN  -- Update account balances based on transaction type
 UPDATE Accounts
    SET Balance = 
        CASE 
            WHEN i.TransactionType = 'Debit' THEN Balance - i.Amount
            WHEN i.TransactionType = 'Credit' THEN Balance + i.Amount
        END
    FROM Accounts a
    INNER JOIN inserted i ON a.AccountID = i.AccountID
    WHERE i.TransactionType IN ('Debit', 'Credit');
END;

----QUESTION 5: Trigger – Audit Transactions (15 Marks)-----

CREATE TRIGGER trg_AuditTransactionInsert
ON Transactions
AFTER INSERT
AS
BEGIN
    -- Log the insert action into the AuditLog table
    INSERT INTO AuditLog (TableName, ActionType, ActionDate, Description)
    SELECT 
   'Transactions',                         -- Table name being affected
        'INSERT',                               -- Action type
        GETDATE(),                             -- Current date and time
        CONCAT('Inserted transaction of R', i.Amount, ' for Account ID ', i.AccountID)  -- Description of the transaction
    FROM inserted i;                          -- Use the 'inserted' table to access the new records
END;

CREATE TRIGGER trg_PreventDelete_Transactions
ON Transactions
INSTEAD OF DELETE
AS
BEGIN      -- Raise an error message to prevent deletion
RAISERROR('Deleting financial transactions is not allowed.', 16, 1);
ROLLBACK;  -- Rollback the delete operation
END;

---QUESTION 7: Trigger – Prevent Negative Balances (10 Marks)

DROP TRIGGER IF EXISTS trg_PreventNegativeBalance;

CREATE TRIGGER trg_PreventNegativeBalance
ON Accounts
INSTEAD OF UPDATE
AS
BEGIN
    -- Check if any updated balances would result in a negative balance
    IF EXISTS (
        SELECT 1
        FROM inserted i
        JOIN Accounts a ON i.AccountID = a.AccountID
        WHERE (a.Balance + (i.Balance - a.Balance) < 0)  -- Check for negative balance
    )
    BEGIN
        -- Raise an error message
        RAISERROR('Account balance cannot go below zero.', 16, 1);
        ROLLBACK;  -- Roll back the update transaction
        RETURN;  -- Exit the trigger after rollback
    END

    -- Proceed with the update
    UPDATE Accounts
    SET Balance = i.Balance
    FROM inserted i
    WHERE Accounts.AccountID = i.AccountID;
END;
