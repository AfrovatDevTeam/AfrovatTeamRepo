    public partial class MainWindow : Window
    {
        public MainWindow()
        {
            InitializeComponent();


            string[] a = { "Celsius", "Fahrenheit", };
            string[] b = { "Fahrenheit", "Celsius", };

            combobox1.ItemsSource = a;
            combobox2.ItemsSource = b;
        }

        private void btnconvert_Click(object sender, RoutedEventArgs e)
        {

            double startingTemp;
            double resultTemp;
            startingTemp = double.Parse(txtinput.Text);


            if (combobox1.Text == "Celsius" && combobox2.Text == "Fahrenheit")
            {
                resultTemp = (startingTemp * (1.8 + 32));
                txtresults.Text = resultTemp.ToString() + "degress in Celsius";
            }
            if (combobox1.Text == "Fahrenheit" && combobox2.Text == "Celsius")
            {
                resultTemp = ((startingTemp - 32) / 1.8);
                txtresults.Text = resultTemp.ToString() + "degress in Fahrenheit";

            }
        }
