using System;
namespace ArrayApplication
{
    class SortArray
    {
        static void Main(string[] args)
        {
            int[] numbers = { 50, 20, 40, 10, 30 };

            Array.Sort(numbers);
            

            Console.WriteLine("Sorted Array: ");
            foreach (int num in numbers)
            {
                Console.WriteLine(num + "");
            }
            Console.ReadKey();
        }
    }
}


//output

Sorted Array:
10
20
30
40
50
