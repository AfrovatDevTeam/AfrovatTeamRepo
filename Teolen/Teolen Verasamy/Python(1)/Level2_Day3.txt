class shapes:
    def area(self):
        return 0
class square(shapes):
    def __init__(self, L, W):
        self.L = L
        self.W = W
    def area(self):
         return self.L*self.W
class circle(shapes):
    def __init__(self, r):
        self.r = r
    def area(self):
         return 3.142*self.r*self.r
s = [square(50,50), circle(25)]
for shapes in s:
    print(shapes.area())

#Encapsulation
class student:
    def __init__(self, name):
        self.__name = name #private variable
    def get_name(self):
        return self.__name
    def set_name(self, new_name):
        self.__name = new_name  #setting method
S = student('Thomas')
print(S.get_name())
S.set_name('Gutter')
print(S.get_name())

#protecting bank info
class Bank_Account:
    def __init__(self, balance):
        self.__balance = balance
    def deposit(self, amount):
        self.__balance += amount
    def get_balance (self):
        return self.__balance
BA = Bank_Account(1000)
BA.deposit(100)
print(BA.get_balance())
