#scatter plot and regression line
import numpy as np
import matplotlib.pyplot as plt
data1 = [1,2,3,4]
data2 = [11,22,33,44]

#
m,b = np.polyfit(data1, data2, 1)
#creating a line across a range
x_line = np.linspace(min(data1), max(data1), 100)
y_line = m*x_line+b
#ploting regression line
plt.plot(x_line, y_line ,color = 'purple', label = f"Line:y={m:.2f}x + {b:.2f}")
#

plt.scatter(data1, data2, color = 'magenta')
plt.legend()

plt.title('Data')
plt.xlabel('data1')
plt.ylabel('data2')
plt.show

class furniture:

        self.price = price

    def total_price(self):
        return self.price*self.quantity_bought
