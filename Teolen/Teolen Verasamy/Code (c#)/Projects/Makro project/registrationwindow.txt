using System;

using System.Text;

using System.Windows.Controls;

namespace MakroPj
{
    /// <summary>
    /// Interaction logic for registrationwindow.xaml
    /// </summary>
    public partial class registrationwindow : Window
    {
        public registrationwindow()
        {
            InitializeComponent();
        }

        private void registerbttn_Click(object sender, RoutedEventArgs e)
        {
            ConnectToDB dbc = new ConnectToDB();

            dbc.RegisterProduct(Convert.ToInt32(IDinput.Text),nameinput.Text,descriptioninput.Text,soldbyinput.Text,Quantityinput.Text,amountinput.Text);
            MessageBox.Show("Product Registered Successfully!");
            IDinput.Text = "";
            nameinput.Text = "";
            descriptioninput.Text = "";
            soldbyinput.Text = ""; 
            Quantityinput.Text = "";
            amountinput.Text = "";

        }

        private void deletebttn_Click(object sender, RoutedEventArgs e)
        {
            ConnectToDB dbc = new ConnectToDB();    
            dbc.deleteProduct(Convert.ToInt32(IDinput.Text));

            MessageBox.Show("Product Deleted Successfully!");
            IDinput.Text = "";           
        }

        private void updatebttn_Click(object sender, RoutedEventArgs e)
        {
            ConnectToDB dbc = new ConnectToDB();

            dbc.UpdateProductInfo(Convert.ToInt32(IDinput.Text), nameinput.Text, descriptioninput.Text, soldbyinput.Text, Quantityinput.Text, amountinput.Text);
            MessageBox.Show("Product Updated Successfully!");

            IDinput.Text = "";
            nameinput.Text = "";
            descriptioninput.Text = "";
            soldbyinput.Text = "";
            Quantityinput.Text = "";
            amountinput.Text = "";
        }
    }
}
