namespace StudentLogin2

    /// <summary>
    /// Interaction logic for studentReg.xaml
    /// </summary>
    public partial class studentReg : Window
    {
        public studentReg()
        {
            InitializeComponent();
        }

        private void btnRegister_Click(object sender, RoutedEventArgs e)
        {
            connecttoDB dbc = new connecttoDB();

            dbc.RegisterStudents(Convert.ToInt32(txtID.Text), txtName.Text, txtSurname.Text, txtAddress.Text, txtRegFees.Text);
            MessageBox.Show("Student Registered Successfully");
            txtID.Text = "";
            txtName.Text = "";
            txtSurname.Text = "";
            txtAddress.Text = "";
            txtRegFees.Text = "";
            


            
        }

        private void Button_Click(object sender, RoutedEventArgs e)
        {
            connecttoDB dbc = new connecttoDB();
            dbc.DeleteStudent(Convert.ToInt32(txtID.Text));
            MessageBox.Show("Customer deleted Successfully");
            txtID.Text = "";
        }

        private void Button_Click_1(object sender, RoutedEventArgs e)
        {
            connecttoDB dbc = new connecttoDB();
            dbc.UpdateStudentInfor(Convert.ToInt32(txtID.Text), txtName.Text, txtSurname.Text, txtAddress.Text, txtRegFees.Text);
            MessageBox.Show("Student Info updated successfully");

            txtID.Text = "";
            txtName.Text = "";
            txtSurname.Text = "";
            txtAddress.Text = "";
            txtRegFees.Text = "";

                
        }
    }
}
