                SqlCommand sqlcmd = new SqlCommand(query, con);
                sqlcmd.CommandType = System.Data.CommandType.Text;
                sqlcmd.Parameters.AddWithValue("@ID", IDinput.Text);
                sqlcmd.Parameters.AddWithValue("@Name", nameinput.Text);
                sqlcmd.Parameters.AddWithValue("@Surname", surnameinput.Text);
                sqlcmd.Parameters.AddWithValue("@Address", addressinput.Text);
                sqlcmd.Parameters.AddWithValue("@RegFees", feesinput.Text);

                int count = Convert.ToInt32(sqlcmd.ExecuteScalar());

                }
            }

            }
            finally
            {
                con.Close();
            }
